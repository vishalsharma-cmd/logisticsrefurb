  function callApi(action, payload) {
    if (!navigator.onLine) return Promise.reject(new Error('No internet connection — try again once you have signal.'));
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT_MS);
    return fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' }, // avoids CORS preflight
      body: JSON.stringify({ action, payload }),
      signal: controller.signal
    }).then(r => r.text()).then(text => {
      clearTimeout(timer);
      let res;
      try { res = JSON.parse(text); }
      catch (e) {
        throw new Error('Server did not return JSON. First 150 chars: ' + text.slice(0, 150));
      }
      if (res && res.error) throw new Error(res.error);
      return res;
    }).catch(err => {
      clearTimeout(timer);
      if (err.name === 'AbortError') throw new Error('Request timed out — weak network, please retry.');
      throw err;
    });
  }

  function attachMic(btnId, input) {
    const btn = document.getElementById(btnId);

      };

    };
  }

  // Compress + resize before converting to base64 — keeps uploads fast on field networks.
  function resizeCanvasToDataUrl(canvas, maxDim, quality) {
    let { width, height } = canvas;
    if (width <= maxDim && height <= maxDim) return canvas.toDataURL('image/jpeg', quality);
    const scale = maxDim / Math.max(width, height);
    const out = document.createElement('canvas');
    out.width = Math.round(width * scale); out.height = Math.round(height * scale);
    out.getContext('2d').drawImage(canvas, 0, 0, out.width, out.height);
    return out.toDataURL('image/jpeg', quality);
  }

  function fileToCompressedBase64(file, maxDim, quality) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      const reader = new FileReader();
      reader.onload = ev => { img.src = ev.target.result; };
      reader.onerror = reject;
      img.onload = () => {
        const canvas = document.createElement('canvas');
        canvas.width = img.width; canvas.height = img.height;
        canvas.getContext('2d').drawImage(img, 0, 0);
        resolve(resizeCanvasToDataUrl(canvas, maxDim, quality));
      };
      img.onerror = reject;
      reader.readAsDataURL(file);
    });
  }

  // Bike side photo — plain native camera capture (fast, no guide needed)
  document.getElementById('sidePhotoBtn').addEventListener('click', () => document.getElementById('sidePhotoInput').click());
  document.getElementById('sidePhotoInput').addEventListener('change', e => {
    const file = e.target.files[0];
    if (!file) return;
    fileToCompressedBase64(file, 1280, 0.72)
      .then(url => setPhoto('sideImage', url, 'sideBox', 'sidePreview', 'sidePhotoBtn', '📷 Retake side photo'))
      .catch(() => showToast('Could not read photo, try again', false));
  });

  // Meter photo — guided in-app camera with a centring circle, falls back
  // to the plain native camera if getUserMedia isn't available.
  let camStream = null;
  const camOverlay = document.getElementById('camOverlay');
  const camVideo = document.getElementById('camVideo');
  const camCanvas = document.getElementById('camCanvas');

  document.getElementById('meterPhotoBtn').addEventListener('click', openMeterCamera);

  function openMeterCamera() {
    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia || !window.isSecureContext) {
      document.getElementById('meterPhotoInput').click();
      return;
    }
    navigator.mediaDevices.getUserMedia({ video: { facingMode: { ideal: 'environment' } }, audio: false })
      .then(stream => {
        camStream = stream;
        camVideo.srcObject = stream;
        camOverlay.classList.remove('hidden');
        sizeGuide();
      })
      .catch(() => document.getElementById('meterPhotoInput').click()); // permission denied etc → fallback
  }

  function sizeGuide() {
    const wrap = document.getElementById('camWrap');
    const guide = document.getElementById('camGuide');
    const size = Math.min(wrap.clientWidth, wrap.clientHeight) * 0.7;
    guide.style.width = size + 'px';
    guide.style.height = size + 'px';
    guide.style.left = (wrap.clientWidth - size) / 2 + 'px';
    guide.style.top = (wrap.clientHeight - size) / 2 + 'px';
  }
  window.addEventListener('resize', () => { if (!camOverlay.classList.contains('hidden')) sizeGuide(); });

  function closeCam() {
    if (camStream) { camStream.getTracks().forEach(t => t.stop()); camStream = null; }
    camOverlay.classList.add('hidden');
  }
  document.getElementById('camCancel').addEventListener('click', closeCam);

  document.getElementById('shutterBtn').addEventListener('click', () => {
    camCanvas.width = camVideo.videoWidth;
    camCanvas.height = camVideo.videoHeight;
    camCanvas.getContext('2d').drawImage(camVideo, 0, 0);
    const dataUrl = resizeCanvasToDataUrl(camCanvas, 1280, 0.72);
    setPhoto('meterImage', dataUrl, 'meterBox', 'meterPreview', 'meterPhotoBtn', '📷 Retake meter photo');
    closeCam();
  });

  // Fallback path (no camera API / permission denied)
  document.getElementById('meterPhotoInput').addEventListener('change', e => {
    const file = e.target.files[0];
    if (!file) return;
    fileToCompressedBase64(file, 1280, 0.72)
      .then(url => setPhoto('meterImage', url, 'meterBox', 'meterPreview', 'meterPhotoBtn', '📷 Retake meter photo'))
      .catch(() => showToast('Could not read photo, try again', false));
  });

  // ---------------- Validate + submit ----------------
  function validate() {
    const ok = navigator.onLine && state.createdBy && state.reg && state.movementType &&
      (state.movementType === 'inward' ? state.from : state.to) &&
      state.sideImage && state.meterImage;
    document.getElementById('submitBtn').disabled = !ok;
  }

  document.getElementById('submitBtn').addEventListener('click', () => {
    const btn = document.getElementById('submitBtn');
    btn.disabled = true; btn.textContent = 'Saving…';
    callApi('submitMovementLog', {
      procId: state.procId, reg: state.reg, make: state.make,
      model: state.model, variant: state.variant,
      movementType: state.movementType, from: state.from, to: state.to,
      sideImageBase64: state.sideImage, sideImageMime: state.sideMime,
      meterImageBase64: state.meterImage, meterImageMime: state.meterMime,
      createdBy: state.createdBy
    }).then(() => {
      showToast(`Saved ✔ ${state.reg} — ${state.movementType}`, true);
      resetForm();
    }).catch(err => {
      showToast(err.message, false);
      btn.disabled = false; btn.textContent = 'Submit Entry / जमा करें';
    });
  });

  function resetForm() {
    Object.assign(state, { reg: '', procId: '', make: '', model: '', variant: '',
      movementType: '', from: '', to: '', sideImage: null, meterImage: null });
    regInput.value = '';
    document.getElementById('fetchedInfo').classList.add('hidden');
    document.getElementById('notFoundMsg').classList.add('hidden');
    btnInward.classList.remove('active'); btnOutward.classList.remove('active');
    fromRow.classList.add('hidden'); toRow.classList.add('hidden');
    fromInput.value = ''; toInput.value = '';
    ['sidePreview', 'meterPreview'].forEach(id => document.getElementById(id).classList.add('hidden'));
    ['sideBox', 'meterBox'].forEach(id => document.getElementById(id).classList.remove('filled'));
    document.getElementById('sidePhotoBtn').textContent = '📷 Take side photo';
    document.getElementById('meterPhotoBtn').textContent = '📷 Take meter photo';
    const btn = document.getElementById('submitBtn');
    btn.textContent = 'Submit Entry / जमा करें'; btn.disabled = true;
  }

  refreshOnlineState();
